1 - Criação do Banco e Tabelas
CREATE DATABASE pousada_db;
USE pousada_db;

CREATE TABLE acomodacoes (
id INT AUTO_INCREMENT NOT NULL PRIMARY KEY,
identificacao VARCHAR(100) NOT NULL,
lotacao INT NOT NULL,
frente_praia BOOLEAN,
valor_diaria DECIMAL(10,2));

CREATE TABLE hospedes (
id INT AUTO_INCREMENT NOT NULL PRIMARY KEY,
nome_completo VARCHAR(255) NOT NULL,
correio_eletronico VARCHAR(255) NOT NULL,
celular VARCHAR(255) NOT NULL);

CREATE TABLE estadias (
id INT AUTO_INCREMENT NOT NULL PRIMARY KEY,
acomodacao_id INT NOT NULL,
hospede_id INT NOT NULL,
checkin DATE NOT NULL,
checkout DATE NOT NULL,

FOREIGN KEY (acomodacao_id) REFERENCES acomodacoes(id),
FOREIGN KEY (hospede_id) REFERENCES hospedes(id));

 3 - Consultas (SELECTs)
SELECT * FROM acomodacoes WHERE frente_praia = 1;

SELECT nome_completo, correio_eletronico FROM hospedes;

SELECT * FROM acomodacoes WHERE lotacao >= 3;

SELECT * FROM hospedes WHERE nome_completo LIKE "M%";

SELECT * FROM estadias WHERE checkin >= '2023-06-01' AND checkin <= '2023-06-30';


 4 - Funções de Agregação
SELECT COUNT(*) FROM acomodacoes;

SELECT MIN(valor_diaria) FROM acomodacoes;

SELECT MAX(valor_diaria) FROM acomodacoes;

SELECT AVG(valor_diaria) FROM acomodacoes;

SELECT COUNT(frente_praia) FROM acomodacoes WHERE frente_praia = true;


 5 - Ordenação
SELECT * FROM acomodacoes ORDER BY valor_diaria ASC;

SELECT * FROM acomodacoes ORDER BY valor_diaria DESC;


 6 - Alteração de Tabela e Atualizações

 7 e 8 - Junções (JOINs)
SELECT * FROM estadias JOIN hospedes ON hospedes.id = estadias.hospede_id;

SELECT * FROM estadias JOIN acomodacoes ON acomodacoes.id = estadias.acomodacao_id;


 9 - Exclusão de Registros
